import queue
import threading
import time


class BroadcastHub:
    def __init__(self):
        self.clients = []
        self.lock = threading.Lock()

    def join(self):
        inbox = queue.Queue()
        with self.lock:
            self.clients.append(inbox)
        return inbox

    def broadcast(self, text):
        with self.lock:
            for inbox in self.clients:
                inbox.put(text)


hub = BroadcastHub()

import time


def reconnect_with_backoff(connect, tries=4):
    delay = 0.1
    for attempt in range(1, tries + 1):
        try:
            return connect()
        except ConnectionError:
            print("attempt", attempt, "failed; waiting", delay)
            time.sleep(delay)
            delay *= 2
    raise ConnectionError("gave up after retries")


def fake_connect():
    raise ConnectionError("network down")


try:
    reconnect_with_backoff(fake_connect)
except ConnectionError as error:
    print("reconnect result:", error)
