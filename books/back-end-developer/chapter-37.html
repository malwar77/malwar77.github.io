import sqlite3


MIGRATIONS = [
    (1, "CREATE TABLE IF NOT EXISTS tasks (id INTEGER PRIMARY KEY, title TEXT)"),
    (2, "ALTER TABLE tasks ADD COLUMN done INTEGER NOT NULL DEFAULT 0"),
]


def apply_migrations(path="app.db"):
    db = sqlite3.connect(path)
    db.execute("CREATE TABLE IF NOT EXISTS schema_version (version INTEGER PRIMARY KEY)")
    done = {row[0] for row in db.execute("SELECT version FROM schema_version")}
    for version, sql in MIGRATIONS:
        if version not in done:
            db.execute(sql)
            db.execute("INSERT INTO schema_version (version) VALUES (?)", (version,))
            print("applied migration", version)
    db.commit()
    db.close()

import sqlite3


def backfill_in_batches(path=":memory:", batch=2):
    db = sqlite3.connect(path)
    db.execute("CREATE TABLE IF NOT EXISTS people (id INTEGER PRIMARY KEY, name TEXT, display TEXT)")
    for i in range(5):
        db.execute("INSERT OR IGNORE INTO people (id, name) VALUES (?, ?)", (i, "user " + str(i)))
    db.commit()
    while True:
        rows = db.execute("SELECT id, name FROM people WHERE display IS NULL LIMIT ?", (batch,)).fetchall()
        if not rows:
            break
        for row_id, name in rows:
            db.execute("UPDATE people SET display = ? WHERE id = ?", (name.strip().title(), row_id))
        db.commit()
        print("backfilled batch of", len(rows))
    db.close()
