import sqlite3


def build_report():
    db = sqlite3.connect("backendday.db")
    total = db.execute("SELECT COUNT(*) FROM tasks").fetchone()[0]
    done = db.execute("SELECT COUNT(*) FROM tasks WHERE done = 1").fetchone()[0]
    db.close()
    lines = ["BackEndDay Daily Report", "-----------------------"]
    lines.append("Tasks in total: " + str(total))
    lines.append("Tasks finished: " + str(done))
    lines.append("Tasks still open: " + str(total - done))
    return "\n".join(lines) + "\n"
