import sqlite3
import queue


class SmallPool:
    def __init__(self, path, size=3):
        self.free = queue.Queue()
        for _ in range(size):
            self.free.put(sqlite3.connect(path, check_same_thread=False))

    def borrow(self):
        return self.free.get()

    def give_back(self, connection):
        self.free.put(connection)


pool = SmallPool(":memory:")
conn = pool.borrow()
conn.execute("CREATE TABLE notes (id INTEGER PRIMARY KEY, text TEXT)")
pool.give_back(conn)
print("pool size stays small and reuse stays simple")
