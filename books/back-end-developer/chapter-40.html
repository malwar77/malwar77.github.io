import json
import urllib.request
import urllib.error


def call_api(url, timeout=5):
    request = urllib.request.Request(url, headers={"Accept": "application/json"})
    try:
        with urllib.request.urlopen(request, timeout=timeout) as response:
            return response.status, json.loads(response.read().decode())
    except urllib.error.HTTPError as error:
        return error.code, {"error": "request failed"}
    except urllib.error.URLError as error:
        return None, {"error": str(error.reason)}


print("client ready: call_api checks status and never waits forever")

import time
import urllib.error


def call_with_retries(work, tries=3):
    delay = 0.2
    for attempt in range(1, tries + 1):
        try:
            return work()
        except urllib.error.HTTPError as error:
            if error.code not in (429, 500, 502, 503) or attempt == tries:
                raise
            print("retryable status", error.code, "- waiting", delay)
            time.sleep(delay)
            delay *= 2


print(call_with_retries(lambda: {"ok": True}))
