from http.server import BaseHTTPRequestHandler, HTTPServer
import json
import threading


def make_handler(name):
    class Handler(BaseHTTPRequestHandler):
        def do_GET(self):
            body = json.dumps({"service": name, "status": "ok"}).encode()
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)

        def log_message(self, *args):
            pass
    return Handler


catalog = HTTPServer(("localhost", 0), make_handler("catalog"))

import time


class CircuitBreaker:
    def __init__(self, limit=3, cool_off=1.0):
        self.limit = limit
        self.cool_off = cool_off
        self.failures = 0
        self.opened_at = None

    def call(self, work):
        if self.opened_at and time.monotonic() - self.opened_at < self.cool_off:
            return "circuit open: fallback used"
        try:
            result = work()
            self.failures = 0
            return result
        except Exception:
            self.failures += 1
            if self.failures >= self.limit:
                self.opened_at = time.monotonic()
            return "fallback used"
