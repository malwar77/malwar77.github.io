import (
	"encoding/json"
	"fmt"
	"net/http"
	"time"
)

type User struct {
	Name string `json:"name"`
}

func main() {
	client := &http.Client{Timeout: 10 * time.Second}

		fmt.Println("error:", err)
		return
	}

		return
