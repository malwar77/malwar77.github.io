// What a computer network really is
#include <iostream>
#include <string>
#include <vector>
std::vector<std::string> split(const std::string& s, char sep) {
    std::vector<std::string> parts;
    std::string cur;
    for (char c : s) { if (c == sep) { parts.push_back(cur); cur.clear(); } else cur += c; }
    parts.push_back(cur);
    return parts;
}
int main() {

    for (const auto& p : parts) std::cout << p << " ";
    std::cout << "\n";
    return 0;
}
